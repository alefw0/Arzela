\section{Enunciado y demostración}

\begin{frame}{Enunciado}
    \begin{teorema}[Arzelà: Convergencia acotada]
        Supongamos que $\set{f_n}$ es una sucesión acotadamente convergente en $[a,b]$ y supongamos que cada $f_n$ es integrable de Riemann en $[a,b]$. Supongamos también que la función límite $f$ es integrable de Riemann en $[a,b]$. Entonces
        \[
            \lim_{n\to\infty}\int_a^b f_n(x)\,dx=\int_a^b\lim_{n\to\infty}f_n(x)\,dx=\int_a^b f(x)\,dx.
        \]
    \end{teorema}
\end{frame}
