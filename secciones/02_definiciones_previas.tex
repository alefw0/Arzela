\section{Definiciones previas}

\begin{frame}{Sucesiones de funciones}
    Sea $f_n:[a,b]\to\R$ para cada $n\in\N$.
    \begin{definicion}
        $\set{f_n}$ \textbf{converge puntualmente} a $f$ si, para cada $x\in[a,b]$,
        \[
            \lim_{n\to\infty}f_n(x)=f(x).
        \]
    \end{definicion}
    \begin{definicion}
        $\set{f_n}$ \textbf{converge uniformemente} a $f$ si
        \[
            \lim_{n\to\infty}\ \sup_{x\in[a,b]}\abs{f_n(x)-f(x)}=0.
        \]
    \end{definicion}
\end{frame}
